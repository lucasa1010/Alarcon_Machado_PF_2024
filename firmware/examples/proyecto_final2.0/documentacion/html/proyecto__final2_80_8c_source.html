141static void controlar(void *pvParameter){
142 while(1){
143 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
144 if (pierde){
145 NeoPixelAllColor(NEOPIXEL_COLOR_RED);
146 GPIOOn(GPIO_BUZZER);
147 vTaskDelay(3000 / portTICK_PERIOD_MS);
148 GPIOOff(GPIO_BUZZER);
149 apagarLeds();
150 conteo_en_ms = 0;
151 medidaAcertada = true;
152 pierde = false;
153 }
154 if(medidaAcertada){ // Si el sensor seleccionado es el correcto...
155 manejoDeLEDs();
156 obtenerSensorPrendido();
157 medidaAcertada = false;
158 }
159 if (deteccionIR){ // Si hay una deteccion del sensor IR...
160 AnalogInputReadSingle(CH1, &voltajeMedido); // Se obtiene el voltaje en el sensor
161
162 if(voltajeMedido > (tensionSensorPrendido - 100) && voltajeMedido < (tensionSensorPrendido + 100)){
163 apagarLeds();
164 medidaAcertada = true;
165 printf("Tiempo medido: %.2f seg\r\n", (float)conteo_en_ms / 1000);
166 conteo_en_ms = 0;
167 }
168 deteccionIR = false;
169 }
170 }
171}
